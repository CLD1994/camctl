function(host_regex_literal output value)
  set(pattern "${value}")
  foreach(character IN ITEMS "\\" "." "^" "$" "*" "+" "?" "(" ")" "[" "]" "|" "{" "}")
    string(REPLACE "${character}" "\\${character}" pattern "${pattern}")
  endforeach()
  set(${output} "${pattern}" PARENT_SCOPE)
endfunction()
