cmake_minimum_required(VERSION 3.10.2)
project(camctl_host VERSION 1.0.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "camctl_host 目前面向 Linux")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(HOST_BUILD_TESTS "构建组件单元与集成测试" OFF)
include(vendor/manifest.cmake)
foreach(entry IN LISTS HOST_VENDOR_FILES)
  string(REPLACE "|" ";" parts "${entry}")
  list(GET parts 0 path)
  list(GET parts 1 expected)
  if(HOST_BUILD_TESTS OR NOT path MATCHES "^cmocka/")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${path}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "供应依赖文件摘要不匹配：${path}")
    endif()
  endif()
endforeach()
find_package(Threads REQUIRED)
add_library(camctl_host STATIC src/result.c src/scheduler.c src/log_queue.c src/process.c src/claim.c
  src/config.c src/input.c src/host.c src/logger.c src/claim_posix.c vendor/yyjson/yyjson.c)
target_include_directories(camctl_host PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> PRIVATE src vendor/yyjson)
target_compile_definitions(camctl_host PRIVATE _GNU_SOURCE)
target_compile_options(camctl_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(camctl_host PUBLIC Threads::Threads)
add_executable(host-demo src/demo.c)
target_compile_definitions(host-demo PRIVATE _GNU_SOURCE)
target_compile_options(host-demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(host-demo PRIVATE camctl_host)
if(HOST_BUILD_TESTS)
  enable_testing()
  add_library(host_cmocka STATIC vendor/cmocka/cmocka.c)
  target_include_directories(host_cmocka PUBLIC vendor/cmocka)
  target_compile_definitions(host_cmocka PRIVATE _GNU_SOURCE HAVE_SIGNAL_H HAVE_STRINGS_H
    HAVE_INTTYPES_H HAVE_STRSIGNAL HAVE_SIGLONGJMP HAVE_VSNPRINTF HAVE_SNPRINTF
    HAVE_GCC_THREAD_LOCAL_STORAGE)
  function(host_unit name)
    add_executable(test_${name} tests/unit/test_${name}.c)
    target_include_directories(test_${name} PRIVATE src)
    target_link_libraries(test_${name} PRIVATE camctl_host host_cmocka)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME unit_${name} COMMAND test_${name})
    set_tests_properties(unit_${name} PROPERTIES LABELS unit TIMEOUT 15)
  endfunction()
  host_unit(result)
  host_unit(scheduler)
  host_unit(log_queue)
  host_unit(claim)
  host_unit(input)
  host_unit(process_unit)
  set_property(TARGET test_process_unit APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--wrap=waitpid,--wrap=read,--wrap=close")
  add_executable(fake_cli tests/integration/fake_cli.c)
  add_executable(test_process tests/integration/test_process.c)
  target_include_directories(test_process PRIVATE src)
  target_link_libraries(test_process PRIVATE camctl_host)
  add_test(NAME integration_process COMMAND test_process $<TARGET_FILE:fake_cli>)
  set_tests_properties(integration_process PROPERTIES LABELS integration TIMEOUT 15)
  add_executable(test_init_failure tests/integration/test_init_failure.c)
  target_link_libraries(test_init_failure PRIVATE camctl_host)
  set_property(TARGET test_init_failure APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--wrap=malloc,--wrap=calloc,--wrap=pipe2,--wrap=pthread_create")
  add_test(NAME integration_init_failure COMMAND test_init_failure)
  set_tests_properties(integration_init_failure PROPERTIES LABELS integration TIMEOUT 15)
  add_executable(test_closed_stdio tests/integration/test_closed_stdio.c)
  target_include_directories(test_closed_stdio PRIVATE src)
  target_link_libraries(test_closed_stdio PRIVATE camctl_host)
  add_test(NAME integration_closed_stdio COMMAND test_closed_stdio)
  set_tests_properties(integration_closed_stdio PROPERTIES LABELS integration TIMEOUT 15)
  add_executable(test_claim_races tests/integration/test_claim_races.c)
  target_include_directories(test_claim_races PRIVATE src)
  target_link_libraries(test_claim_races PRIVATE camctl_host)
  set_property(TARGET test_claim_races APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--wrap=renameat")
  foreach(mode RANGE 1 3)
    add_test(NAME integration_claim_race_${mode} COMMAND test_claim_races ${mode})
    set_tests_properties(integration_claim_race_${mode} PROPERTIES LABELS integration TIMEOUT 15)
  endforeach()
  find_program(HOST_TEST_PYTHON NAMES python3)
  if(NOT HOST_TEST_PYTHON)
    message(FATAL_ERROR "集成测试需要 Python 3")
  endif()
  add_executable(host_driver tests/integration/host_driver.c)
  target_link_libraries(host_driver PRIVATE camctl_host)
  set_property(TARGET host_driver APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--wrap=write,--wrap=clock_gettime")
  add_test(NAME integration_host COMMAND ${HOST_TEST_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_host.py
    $<TARGET_FILE:host_driver> ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/controlled_cli.py)
  set_tests_properties(integration_host PROPERTIES LABELS integration TIMEOUT 90)
  add_test(NAME integration_demo COMMAND ${HOST_TEST_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_demo.py
    $<TARGET_FILE:host-demo> ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/controlled_cli.py)
  set_tests_properties(integration_demo PROPERTIES LABELS integration TIMEOUT 30)
  # Release 测试同样执行 assert 内的协作调用，不能由 NDEBUG 移除验收步骤。
  get_property(host_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS host_targets)
    get_target_property(kind ${target} TYPE)
    if(kind STREQUAL "EXECUTABLE" AND NOT target STREQUAL "host-demo")
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
include(GNUInstallDirs)
install(TARGETS camctl_host ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS host-demo RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/camctl_host.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES vendor/yyjson/LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR}/yyjson)
install(FILES src/demo.c DESTINATION ${CMAKE_INSTALL_DOCDIR}/examples)
set(CPACK_GENERATOR "TGZ")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_IGNORE_FILES "/[.]git/;/[.]local/;/build[^/]*/;__pycache__;[.]pyc$")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CPack)
endif()
